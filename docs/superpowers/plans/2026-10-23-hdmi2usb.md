> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把安卓平板（Android 16+）变成 PC 扩展副屏：全屏低延迟显示 UVC 采集卡画面，插线自启、拔出退出、双击切换缩放、屏幕常亮。

**Tech Stack:** Kotlin 2.1.0、Gradle 8.11.1、AGP 8.9.2、minSdk/targetSdk 36、JUnit 4.13.2（仅测试依赖）、Java 17。

```kotlin
package com.hdmi2usb.monitor

import android.graphics.ImageFormat

/** 候选流排序：MJPEG 优先 → 分辨率不超上限取最大 → 同分辨率取最高帧率。 */
object StreamSelector {

    fun select(
        options: List<StreamOption>,
        maxWidth: Int,
        maxHeight: Int,
    ): List<StreamOption> {
        val valid = options.filter { it.width > 0 && it.height > 0 }
        val inBounds = valid.filter { it.width <= maxWidth && it.height <= maxHeight }
        return (inBounds.ifEmpty { valid }).sortedWith(
            compareByDescending<StreamOption> { it.format == ImageFormat.MJPEG }
                .thenByDescending { it.width * it.height }
                .thenByDescending { it.maxFps },
        )
    }
}
```

```kotlin
package com.hdmi2usb.monitor

/** 显示视口计算：FIT 等比留黑边，STRETCH 拉伸铺满。 */
object ScreenFit {

    enum class Mode { FIT, STRETCH }

    data class Viewport(val width: Int, val height: Int, val offsetX: Int, val offsetY: Int)

    fun viewport(
        streamW: Int,
        streamH: Int,
        containerW: Int,
        containerH: Int,
        mode: Mode,
    ): Viewport {
        require(streamW > 0 && streamH > 0 && containerW > 0 && containerH > 0) {
            "stream and container sizes must be positive"
        }
        if (mode == Mode.STRETCH) return Viewport(containerW, containerH, 0, 0)
        val streamRatio = streamW.toDouble() / streamH
        val containerRatio = containerW.toDouble() / containerH
        val (w, h) = if (streamRatio > containerRatio) {
            containerW to Math.round(containerW / streamRatio).toInt()
        } else {
            Math.round(containerH * streamRatio).toInt() to containerH
        }
        return Viewport(w, h, (containerW - w) / 2, (containerH - h) / 2)
    }
}
```

```kotlin
package com.hdmi2usb.monitor

import android.content.Context
import android.graphics.Color
import android.util.AttributeSet
import android.view.GestureDetector
import android.view.MotionEvent
import android.view.SurfaceView
import android.widget.FrameLayout

/**
 * 黑底 SurfaceView 容器。
 * FIT：SurfaceView 按流宽高比定尺寸并居中（等比留黑边）；
 * STRETCH：SurfaceView 铺满容器（拉伸填充）。
 * 双击切换模式，只重布局，不重建相机会话。
 */
class DisplayView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
) : FrameLayout(context, attrs) {

    val surfaceView = SurfaceView(context).also { addView(it) }

    var mode: ScreenFit.Mode = ScreenFit.Mode.FIT
        private set

    var onModeChanged: ((ScreenFit.Mode) -> Unit)? = null

    private var streamW = 0
    private var streamH = 0

    private val gestureDetector = GestureDetector(
        context,
        object : GestureDetector.SimpleOnGestureListener() {
            override fun onDoubleTap(e: MotionEvent): Boolean {
                toggleMode()
                return true
            }
        },
    )

    init {
        setBackgroundColor(Color.BLACK)
    }

    override fun onTouchEvent(event: MotionEvent): Boolean {
        gestureDetector.onTouchEvent(event)
        return true
    }

    /** 相机流就绪：记录分辨率并应用当前缩放模式。 */
    fun onStreamConfigured(width: Int, height: Int) {
        streamW = width
        streamH = height
        applyLayout()
    }

    fun toggleMode() {
        mode = if (mode == ScreenFit.Mode.FIT) ScreenFit.Mode.STRETCH else ScreenFit.Mode.FIT
        applyLayout()
        onModeChanged?.invoke(mode)
    }

    override fun onSizeChanged(w: Int, h: Int, oldw: Int, oldh: Int) {
        super.onSizeChanged(w, h, oldw, oldh)
        applyLayout()
    }

    private fun applyLayout() {
        if (streamW == 0 || streamH == 0 || width == 0 || height == 0) return
        val vp = ScreenFit.viewport(streamW, streamH, width, height, mode)
        val lp = surfaceView.layoutParams as LayoutParams
        lp.width = vp.width
        lp.height = vp.height
        lp.leftMargin = vp.offsetX
        lp.topMargin = vp.offsetY
        surfaceView.layoutParams = lp
    }
}
```

```kotlin
import android.os.Looper
import android.util.Range
import android.view.SurfaceHolder

interface CameraListener {
    /** 流已就绪并开始渲染（主线程回调）。 */
    fun onStreamReady(width: Int, height: Int)
    /** 相机不可用，应提示并退出（主线程回调）。 */
    fun onUnavailable()
}

class CameraController(context: Context, private val listener: CameraListener) {

    private val cameraManager =
        context.getSystemService(Context.CAMERA_SERVICE) as CameraManager
    private val cameraThread = HandlerThread("CameraThread").apply { start() }
    private val cameraHandler = Handler(cameraThread.looper)
    private val mainHandler = Handler(Looper.getMainLooper())

    private var cameraId: String? = null
    private var holder: SurfaceHolder? = null
    private var candidates: List<StreamOption> = emptyList()
    private var candidateIndex = 0
    private var device: CameraDevice? = null
    private var session: CameraCaptureSession? = null
    private var reopenedOnce = false
    private var released = false

    fun findExternalCamera(): String? =
        cameraManager.cameraIdList.firstOrNull { id ->
            cameraManager.getCameraCharacteristics(id)
                .get(CameraCharacteristics.LENS_FACING) == CameraCharacteristics.LENS_FACING_EXTERNAL
        }

    /** 枚举外部相机候选流：MJPEG 优先、不超屏幕、按分辨率/帧率降序。 */
    fun enumerateStreams(id: String, screenW: Int, screenH: Int): List<StreamOption> {
        val map = cameraManager.getCameraCharacteristics(id)
            .get(CameraCharacteristics.SCALER_STREAM_CONFIGURATION_MAP)
            ?: return emptyList()
        val options = mutableListOf<StreamOption>()
        for (format in SUPPORTED_FORMATS) {
            val sizes = map.getOutputSizes(format) ?: continue
            for (size in sizes) {
                val minNs = map.getOutputMinFrameDuration(format, size) // 最短帧间隔 → 最高帧率
                val maxNs = map.getOutputMaxFrameDuration(format, size) // 最长帧间隔 → 最低帧率
                val fpsMax = if (minNs > 0) (1_000_000_000.0 / minNs).toInt() else 0
                val fpsMin = if (maxNs > 0) (1_000_000_000.0 / maxNs).toInt() else fpsMax
                options += StreamOption(format, size.width, size.height, fpsMin, fpsMax)
            }
        }
        return StreamSelector.select(options, screenW, screenH)
    }

    fun start(id: String, holder: SurfaceHolder) {
        cameraId = id
        this.holder = holder
        val frame = holder.surfaceFrame
        candidates = enumerateStreams(id, frame.width(), frame.height())
        if (candidates.isEmpty()) {
            mainHandler.post { listener.onUnavailable() }
            return
        }
        candidateIndex = 0
        reopenedOnce = false
        openCamera()
    }

    @SuppressLint("MissingPermission")
    private fun openCamera() {
        val id = cameraId ?: return
        try {
            cameraManager.openCamera(id, object : CameraDevice.StateCallback() {
                override fun onOpened(camera: CameraDevice) {
                    if (released) {
                        camera.close()
                        return
                    }
                    device = camera
                    createSession()
                }

                override fun onDisconnected(camera: CameraDevice) {
                    camera.close()
                    session = null
                    if (released) return
                    if (!reopenedOnce) {
                        reopenedOnce = true
                        openCamera()
                    } else {
                        mainHandler.post { listener.onUnavailable() }
                    }
                }

                override fun onError(camera: CameraDevice, error: Int) {
                    camera.close()
                    if (released) return
                    mainHandler.post { listener.onUnavailable() }
                }
            }, cameraHandler)
        } catch (_: SecurityException) {
            mainHandler.post { listener.onUnavailable() }
        } catch (_: CameraAccessException) {
            mainHandler.post { listener.onUnavailable() }
        }
    }

    private fun createSession() {
        val camera = device ?: return
        val surface = holder?.surface ?: return
        if (candidateIndex >= candidates.size) {
            mainHandler.post { listener.onUnavailable() }
            return
        }
        val candidate = candidates[candidateIndex]
        try {
            camera.createCaptureSession(listOf(surface), object : CameraCaptureSession.StateCallback() {
                override fun onConfigured(s: CameraCaptureSession) {
                    if (released) {
                        s.close()
                        return
                    }
                    session = s
                    try {
                        val request = camera.createCaptureRequest(CameraDevice.TEMPLATE_PREVIEW).apply {
                            addTarget(surface)
                            if (candidate.maxFps > 0) {
                                set(
                                    CaptureRequest.CONTROL_AE_TARGET_FPS_RANGE,
                                    Range(candidate.minFps.coerceIn(1, candidate.maxFps), candidate.maxFps),
                                )
                            }
                        }
                        s.setRepeatingRequest(request.build(), null, cameraHandler)
                        mainHandler.post { listener.onStreamReady(candidate.width, candidate.height) }
                    } catch (_: Exception) {
                        advanceCandidate()
                    }
                }

                override fun onConfigureFailed(s: CameraCaptureSession) {
                    s.close()
                    if (released) return
                    advanceCandidate()
                }
            }, cameraHandler)
        } catch (_: Exception) {
            advanceCandidate()
        }
    }

    /** 流协商失败 → 逐级降级：换下一个候选配置重建会话。 */
    private fun advanceCandidate() {
        session?.close()
        session = null
        candidateIndex++
        if (candidateIndex >= candidates.size) {
            mainHandler.post { listener.onUnavailable() }
        } else {
            createSession()
        }
    }

    fun close() {
        if (released) return
        released = true
        try {
            session?.close()
        } catch (_: Exception) {
        }
        try {
            device?.close()
        } catch (_: Exception) {
        }
        session = null
        device = null
        cameraThread.quitSafely()
    }

    private companion object {
        /** 采集卡常见输出格式，MJPEG 优先探测。 */
        val SUPPORTED_FORMATS =
            intArrayOf(ImageFormat.MJPEG, ImageFormat.YUV_420_888, ImageFormat.PRIVATE)
    }
}
```

```kotlin
package com.hdmi2usb.monitor

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.content.IntentFilter
import android.hardware.usb.UsbConstants
import android.hardware.usb.UsbDevice
import android.hardware.usb.UsbManager

/** 监听采集卡拔出；插入场景由 manifest 的 USB_DEVICE_ATTACHED intent-filter 承担。 */
class PlugMonitor(
    private val context: Context,
    private val onDetached: () -> Unit,
) {
    private var registered = false

    private val receiver = object : BroadcastReceiver() {
        override fun onReceive(ctx: Context, intent: Intent) {
            if (intent.action != UsbManager.ACTION_USB_DEVICE_DETACHED) return
            val device = intent.getParcelableExtra(UsbManager.EXTRA_DEVICE, UsbDevice::class.java)
            if (device != null && device.isVideoDevice()) onDetached()
        }
    }

    private val filter = IntentFilter(UsbManager.ACTION_USB_DEVICE_DETACHED)

    fun register() {
        if (registered) return
        context.registerReceiver(receiver, filter)
        registered = true
    }

    fun unregister() {
        if (!registered) return
        context.unregisterReceiver(receiver)
        registered = false
    }

    private fun UsbDevice.isVideoDevice(): Boolean {
        // 视频控制类（0x0E）或 IAD 封装的摄像头（0xEF / subclass 2）
        if (deviceClass == UsbConstants.USB_CLASS_VIDEO) return true
        if (deviceClass == 0xEF && deviceSubclass == 2) return true
        for (i in 0 until interfaceCount) {
            if (getInterface(i).interfaceClass == UsbConstants.USB_CLASS_VIDEO) return true
        }
        return false
    }
}
```

```kotlin
package com.hdmi2usb.monitor

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.pm.PackageManager
import android.hardware.usb.UsbManager
import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.view.SurfaceHolder
import android.view.View
import android.view.WindowInsets
import android.view.WindowInsetsController
import android.view.WindowManager
import android.widget.Button
import android.widget.TextView
import android.widget.Toast

class MainActivity : Activity(), SurfaceHolder.Callback, CameraListener {

    private lateinit var displayView: DisplayView
    private lateinit var statusText: TextView
    private lateinit var permissionPanel: View
    private lateinit var permissionText: TextView
    private lateinit var permissionButton: Button

    private var controller: CameraController? = null
    private var plugMonitor: PlugMonitor? = null
    private var surfaceReady = false
    private var retryCount = 0
    private val mainHandler = Handler(Looper.getMainLooper())

    private val reopenRunnable = Runnable { tryOpenCamera() }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        window.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
        window.insetsController?.apply {
            hide(WindowInsets.Type.systemBars())
            systemBarsBehavior = WindowInsetsController.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        }
        setContentView(R.layout.activity_main)
        displayView = findViewById(R.id.display)
        statusText = findViewById(R.id.status_text)
        permissionPanel = findViewById(R.id.permission_panel)
        permissionText = findViewById(R.id.permission_text)
        permissionButton = findViewById(R.id.btn_grant)
        displayView.surfaceView.holder.addCallback(this)
        permissionButton.setOnClickListener {
            requestPermissions(arrayOf(Manifest.permission.CAMERA), REQ_CAMERA)
        }
        if (checkSelfPermission(Manifest.permission.CAMERA) == PackageManager.PERMISSION_GRANTED) {
            startPipeline()
        } else {
            requestPermissions(arrayOf(Manifest.permission.CAMERA), REQ_CAMERA)
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        // singleTask：app 存活时再次插入采集卡，重置重试并重新打开
        if (intent.action == UsbManager.ACTION_USB_DEVICE_ATTACHED && surfaceReady) {
            retryCount = 0
            tryOpenCamera()
        }
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray,
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode != REQ_CAMERA) return
        if (grantResults.firstOrNull() == PackageManager.PERMISSION_GRANTED) {
            permissionPanel.visibility = View.GONE
            startPipeline()
        } else {
            permissionText.setText(R.string.permission_denied)
            permissionButton.setText(R.string.retry_grant)
            permissionPanel.visibility = View.VISIBLE
        }
    }

    private fun startPipeline() {
        if (plugMonitor == null) plugMonitor = PlugMonitor(this) { onCaptureCardDetached() }
        plugMonitor?.register()
        if (controller == null) controller = CameraController(this, this)
        if (surfaceReady) tryOpenCamera()
    }

    private fun tryOpenCamera() {
        val c = controller ?: return
        val id = c.findExternalCamera()
        if (id == null) {
            if (retryCount < MAX_RETRIES) {
                retryCount++
                statusText.setText(R.string.no_capture_card)
                statusText.visibility = View.VISIBLE
                mainHandler.postDelayed(reopenRunnable, RETRY_INTERVAL_MS)
            } else {
                Toast.makeText(this, R.string.no_capture_card_exit, Toast.LENGTH_SHORT).show()
                finish()
            }
        } else {
            statusText.visibility = View.GONE
            c.start(id, displayView.surfaceView.holder)
        }
    }

    override fun surfaceCreated(holder: SurfaceHolder) {
        surfaceReady = true
        if (controller != null) tryOpenCamera()
    }

    override fun surfaceChanged(holder: SurfaceHolder, format: Int, width: Int, height: Int) {
        // 旋转/缩放只改变 surface 尺寸，相机输出自动缩放，无需重建会话
    }

    override fun surfaceDestroyed(holder: SurfaceHolder) {
        surfaceReady = false
    }

    override fun onStreamReady(width: Int, height: Int) {
        displayView.onStreamConfigured(width, height)
    }

    override fun onUnavailable() {
        Toast.makeText(this, R.string.camera_unavailable, Toast.LENGTH_SHORT).show()
        finish()
    }

    private fun onCaptureCardDetached() {
        controller?.close()
        finish()
    }

    override fun onDestroy() {
        super.onDestroy()
        mainHandler.removeCallbacks(reopenRunnable)
        plugMonitor?.unregister()
        controller?.close()
    }

    private companion object {
        const val REQ_CAMERA = 1
        const val MAX_RETRIES = 3
        const val RETRY_INTERVAL_MS = 2_000L
    }
}
```
